cmake_minimum_required(VERSION 3.22.1)
project(pocketnode)

set(NODE_VERSION "18.20.4")
set(NODE_ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/nodejs-mobile-v${NODE_VERSION}-android.zip")
set(NODE_EXTRACT_ROOT "${CMAKE_CURRENT_BINARY_DIR}/node-mobile")
set(NODE_LIB "")
file(GLOB_RECURSE NODE_LIBS "${NODE_EXTRACT_ROOT}/bin/${ANDROID_ABI}/libnode.so" "${NODE_EXTRACT_ROOT}/*/bin/${ANDROID_ABI}/libnode.so")
if(NOT NODE_LIBS)
    file(DOWNLOAD
        "https://github.com/nodejs-mobile/nodejs-mobile/releases/download/v${NODE_VERSION}/nodejs-mobile-v${NODE_VERSION}-android.zip"
        "${NODE_ARCHIVE}" SHOW_PROGRESS TLS_VERIFY ON STATUS DOWNLOAD_STATUS)
    list(GET DOWNLOAD_STATUS 0 DOWNLOAD_CODE)
    if(NOT DOWNLOAD_CODE EQUAL 0)
        message(FATAL_ERROR "Could not download the pinned Node.js mobile runtime: ${DOWNLOAD_STATUS}")
    endif()
    file(ARCHIVE_EXTRACT INPUT "${NODE_ARCHIVE}" DESTINATION "${NODE_EXTRACT_ROOT}")
    file(GLOB_RECURSE NODE_LIBS "${NODE_EXTRACT_ROOT}/bin/${ANDROID_ABI}/libnode.so" "${NODE_EXTRACT_ROOT}/*/bin/${ANDROID_ABI}/libnode.so")
endif()
if(NOT NODE_LIBS)
    message(FATAL_ERROR "Node.js mobile release is missing ${ANDROID_ABI}/libnode.so")
endif()
list(GET NODE_LIBS 0 NODE_LIB)
get_filename_component(NODE_ABI_DIR "${NODE_LIB}" DIRECTORY)
get_filename_component(NODE_BIN_DIR "${NODE_ABI_DIR}" DIRECTORY)
get_filename_component(NODE_ROOT "${NODE_BIN_DIR}" DIRECTORY)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}")
configure_file("${NODE_LIB}" "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libnode.so" COPYONLY)

add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_LIB}")
add_library(pocketnode SHARED node-runtime.cpp)
target_include_directories(pocketnode PRIVATE "${NODE_ROOT}/include/node")
find_library(log-lib log)
target_link_libraries(pocketnode node ${log-lib})
